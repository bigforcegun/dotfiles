# Time Machine — исключения (macOS)
#
# Формат: одна запись на строку, `~` = $HOME, `#` — комментарий.
# Префикс `-p ` — исключение по пути (SkipPaths в /Library/Preferences/com.apple.TimeMachine),
#   ставится через sudo:   sudo tmutil addexclusion -p <path>
# Без префикса — «липкое» исключение (xattr com_apple_backup_excludeItem на самой папке),
#   ставится без sudo:     tmutil addexclusion <path>
#   Живёт на папке: пропадает, если папку удалить и создать заново, и при переустановке системы.
#
# Проверка:   tmutil isexcluded <path>
# Все SkipPaths: defaults read /Library/Preferences/com.apple.TimeMachine SkipPaths
#
# Уже исключено самой системой (не дублировать): ~/Library/Caches, ~/Library/Logs, ~/.cache,
# OrbStack data.img.raw / swap.img (ставит OrbStack).
#
# Собрано 2026-10-09.
